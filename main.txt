int calcularPromedio(int suma, int total) {
    int resultado
    resultado = suma / total
    return resultado
}

int main() {
    int dia
    int contador
    int limite
    int i
    float pi
    float resultado
    string nombre

    pi = 3.14
    contador = 0

    print("Ingrese su nombre:")
    read(nombre)
    print("Hola", nombre)

    print("Introduce un numero de dia (1-3):")
    read(dia)

    switch (dia) {
        case 1:
            print("Lunes")
            break;
        case 2:
            print("Martes")
            break;
        case 3:
            print("Miercoles")
            break;
        default:
            print("Dia no valido")
            break;
    }

    print("Ingrese un limite para el while:")
    read(limite)

    while (contador < limite) {
        print("Contador:", contador)
        contador = contador + 1
    }

    do {
        print("Ejecutando do-while:", contador)
        contador = contador - 1
    } while (contador > 0)

    print("Ciclo for:")
    for (i = 0; i < 5; i = i + 1) {
        print("i:", i)
    }

    if (limite > 5) {
        print("Limite mayor a 5")
    } else {
        print("Limite menor o igual a 5")
    }

    resultado = calcularPromedio(10 + 5 * 2, 3)
    print("Resultado promedio:", resultado)

    print("Valor de PI:", pi)

    return 0
}
