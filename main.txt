int main() {
    // --- 1. DECLARACION DE VARIABLES ---
    int a
    int b
    float x
    float y
    int resInt
    float resFloat
    int opcion
    int control
    int i
    bool valid

    print("--- INICIO DEL PROGRAMA ---")

    // --- 2. ASIGNACION Y LECTURA DESDE TECLADO ---
    print("Ingrese un numero entero para a:")
    read(a)
    print("Ingrese un numero decimal para x:")
    read(x)

    // 3. ASIGNACION DE VALORES A ID (Variables y constantes)
    b = 10
    y = 5.0
    valid = true

    // 4. RESOLVER EXPRESIONES MATEMATICAS (Suma, Resta, Mult, Div)
    // Operacion entera: (a + 10) * 2 / 2
    resInt = a + b
    resInt = resInt * 2
    resInt = resInt / 2
    
    // Operacion decimal: (x + 5.0) - 2.0
    resFloat = x + y
    resFloat = resFloat - 2.0

    // 5. IMPRESION DE RESULTADOS DE ASIGNACIONES Y EXPRESIONES
    print("Resultado de expresion entera calculada:", resInt)
    print("Resultado de expresion decimal calculada:", resFloat)

    // 6. USO DE SENTENCIAS DE CONTROL Y SUS RESULTADOS
    print("--- Evaluando IF-ELSE ---")
    if (resInt > 10) {
        print("El resultado es mayor a 10")
    } else {
        print("El resultado es 10 o menor")
    }

    // --- 6. ESTRUCTURA SWITCH-CASE (Break requiere ;) ---
    print("MENU: 1. Sumar 2. Restar. Elija:")
    read(opcion)
    switch (opcion) {
        case 1:
            print("Suma de a y b:", a + b)
            break;
        case 2:
            print("Resta de a y b:", a - b)
            break;
        default:
            print("Opcion no valida")
            break;
    }

    // --- 7. BUCLES: WHILE, DO-WHILE, FOR ---
    print("Contando con WHILE:")
    control = 0
    while (control < 2) {
        print("While val:", control)
        control = control + 1
    }

    print("Ejecutando DO-WHILE:")
    do {
        print("Do-While val:", control)
        control = control + 1
    } while (control < 3)

    print("Contando con FOR:")
    for (i = 0; i < 3; i = i + 1) {
        print("For index:", i)
    }

    print("--- FIN DEL PROGRAMA ---")
    return 0
}
